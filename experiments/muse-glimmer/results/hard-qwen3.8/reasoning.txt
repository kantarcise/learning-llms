import math
from collections import OrderedDict


class ExpiringLRU:
    def __init__(self, capacity):
        if isinstance(capacity, bool) or not isinstance(capacity, int) or capacity < 0:
            raise ValueError("capacity must be an int >= 0")
        self._capacity = capacity
        self._store = OrderedDict()  # key -> (value, deadline)
        self._last_time = None

    @staticmethod
    def _validate_now(now):
        if isinstance(now, bool) or not isinstance(now, (int, float)):
            raise ValueError("now must be a finite nonnegative int or float")
        if not math.isfinite(now) or now < 0:
            raise ValueError("now must be a finite nonnegative int or float")

    @staticmethod
    def _validate_ttl(ttl):
        if ttl is None:
            return
        if isinstance(ttl, bool) or not isinstance(ttl, (int, float)):
            raise ValueError("ttl must be None or a finite nonnegative int or float")
        if not math.isfinite(ttl) or ttl < 0:
            raise ValueError("ttl must be None or a finite nonnegative int or float")

    @staticmethod
    def _validate_key(key):
        if not isinstance(key, str):
            raise ValueError("key must be a string")

    def _check_time(self, now):
        if self._last_time is not None and now < self._last_time:
            raise ValueError("time must be nondecreasing")

    def _purge(self, now):
        to_remove = []
        for key, (value, deadline) in self._store.items():
            if deadline is not None and deadline <= now:
                to_remove.append(key)
        for key in to_remove:
            del self._store[key]

    def _apply_put(self, key, value, ttl, now):
        """Apply a single put to the internal store (assumes validation done)."""
        if ttl == 0:
            self._store.pop(key, None)
            return
        if self._capacity == 0:
            return
        deadline = None if ttl is None else now + ttl
        if key in self._store:
            del self._store[key]
        self._store[key] = (value, deadline)
        # OrderedDict: insertion at end = MRU

    def _enforce_capacity(self):
        while len(self._store) > self._capacity:
            self._store.popitem(last=False)

    def put(self, key, value, *, now, ttl=None):
        self._validate_key(key)
        self._validate_now(now)
        self._validate_ttl(ttl)
        if ttl is not None:
            deadline = now + ttl
            if not math.isfinite(deadline):
                raise ValueError("now + ttl must be finite")
        self._check_time(now)
        
        self._purge(now)
        self._apply_put(key, value, ttl, now)
        self._enforce_capacity()
        self._last_time = now

    def put_many(self, entries, *, now):
        self._validate_now(now)
        self._check_time(now)
        
        # Consume and validate the entire iterable
        batch = []
        for entry in entries:
            # Must be a triple
            if not isinstance(entry, (tuple, list)) or len(entry) != 3:
                raise ValueError("each entry must be a (key, value, ttl) triple")
            key, value, ttl = entry
            self._validate_key(key)
            self._validate_ttl(ttl)
            if ttl is not None:
                deadline = now + ttl
                if not math.isfinite(deadline):
                    raise ValueError("now + ttl must be finite")
            batch.append((key, value, ttl))
        
        # Now apply atomically
        self._purge(now)
        for key, value, ttl in batch:
            self._apply_put(key, value, ttl, now)
        self._enforce_capacity()
        self._last_time = now

    def get(self, key, default=None, *, now):
        self._validate_key(key)
        self._validate_now(now)
        self._check_time(now)
        
        self._purge(now)
        
        if key in self._store:
            value, deadline = self._store.pop(key)
            self._store[key] = (value, deadline)
            self._last_time = now
            return value
        else:
            self._last_time = now
            return default

    def snapshot(self, *, now):
        self._validate_now(now)
        self._check_time(now)
        
        self._purge(now)
        
        result = [(key, value) for key, (value, deadline) in self._store.items()]
        self._last_time = now
        return result

class ExpiringLRU:
    def __init__(self, capacity):
        if isinstance(capacity, bool) or not isinstance(capacity, int) or capacity < 0:
            raise ValueError("capacity must be an int >= 0")
        self._capacity = capacity
        self._store = OrderedDict()
        self._last_time = None

    @staticmethod
    def _validate_now(now):
        if isinstance(now, bool) or not isinstance(now, (int, float)):
            raise ValueError("now must be a finite nonnegative int or float")
        if not math.isfinite(now) or now < 0:
            raise ValueError("now must be a finite nonnegative int or float")

    @staticmethod
    def _validate_ttl(ttl):
        if ttl is None:
            return
        if isinstance(ttl, bool) or not isinstance(ttl, (int, float)):
            raise ValueError("ttl must be None or a finite nonnegative int or float")
        if not math.isfinite(ttl) or ttl < 0:
            raise ValueError("ttl must be None or a finite nonnegative int or float")

    @staticmethod
    def _validate_key(key):
        if not isinstance(key, str):
            raise ValueError("key must be a string")

    def _check_time(self, now):
        if self._last_time is not None and now < self._last_time:
            raise ValueError("time must be nondecreasing")

    def _purge(self, now):
        to_remove = [k for k, (_, d) in self._store.items() if d is not None and d <= now]
        for k in to_remove:
            del self._store[k]

    def _apply_put(self, key, value, ttl, now):
        if ttl == 0:
            self._store.pop(key, None)
            return
        if self._capacity == 0:
            return

        if key in self._store:
            self._store[key] = (value, deadline)
            self._store.move_to_end(key)
        else:
            self._store[key] = (value, deadline)

    def _enforce_capacity(self):
        while len(self._store) > self._capacity:
            self._store.popitem(last=False)

        self._validate_now(now)
        self._validate_ttl(ttl)
